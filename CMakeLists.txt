cmake_minimum_required(VERSION 3.20)
project(kflib-estimators VERSION 0.0 LANGUAGES CXX C)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD_REQUIRED True)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} -O0 -g -Wall -Wextra -Werror -fPIC")
set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} -O3 -DNDEBUG -w -fPIC")

set(PDIR ${PROJECT_SOURCE_DIR})
set(DIR  ${CMAKE_CURRENT_LIST_DIR})

set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${PDIR}/lib/external)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${PDIR}/lib/external)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${PDIR}/bin/external)
add_subdirectory(external/kflib)

set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${PDIR}/lib/core)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${PDIR}/lib/core)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${PDIR}/bin/core)
add_subdirectory(core)
